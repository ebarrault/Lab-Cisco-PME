! =====================================================================
!  R-AGENCE — routeur Cisco 2911 de l'agence (Pau)
!  Rôles : routage inter-VLAN "router-on-a-stick" (sous-interfaces 802.1Q),
!          DHCP, NAT/PAT, VPN GRE over IPsec vers le siège, OSPF zone 0
!  G0/0 -> ISP G0/1 | G0/1 -> SW-AGENCE Gi0/1 (trunk)
!  Prérequis IPsec : license boot module c2900 technology-package securityk9 + reload
! =====================================================================
enable
configure terminal
hostname R-AGENCE
no ip domain-lookup
ip domain-name pme.local

service password-encryption
security passwords min-length 10
enable secret Cisco#Lab2026!
username admin privilege 15 secret Admin#Lab2026!
login block-for 120 attempts 3 within 60
banner motd # Acces reserve au personnel autorise. Toute connexion est journalisee. #
no ip http server
logging buffered 16384
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
ip access-list standard ACL-ADMIN-SSH
 permit 10.10.99.0 0.0.0.255
 permit 10.20.99.0 0.0.0.255
 deny any log
line console 0
 login local
 exec-timeout 5 0
 logging synchronous
line vty 0 4
 login local
 transport input ssh
 access-class ACL-ADMIN-SSH in
 exec-timeout 5 0
line vty 5 15
 transport input none

interface GigabitEthernet0/0
 description WAN - vers ISP
 ip address 198.51.100.2 255.255.255.252
 ip nat outside
 no cdp enable
 no shutdown
interface GigabitEthernet0/1
 description Trunk vers SW-AGENCE
 no ip address
 no shutdown
interface GigabitEthernet0/1.10
 description VLAN 10 ADMINISTRATIF
 encapsulation dot1Q 10
 ip address 10.20.10.1 255.255.255.0
 ip nat inside
interface GigabitEthernet0/1.99
 description VLAN 99 MANAGEMENT
 encapsulation dot1Q 99
 ip address 10.20.99.1 255.255.255.0
 ip nat inside
interface GigabitEthernet0/1.999
 description VLAN natif (aucun trafic)
 encapsulation dot1Q 999 native

ip dhcp excluded-address 10.20.10.1 10.20.10.20
ip dhcp pool AGENCE-VLAN10
 network 10.20.10.0 255.255.255.0
 default-router 10.20.10.1
 dns-server 10.10.30.10
 domain-name pme.local

ip access-list extended ACL-NAT
 deny ip 10.0.0.0 0.255.255.255 10.0.0.0 0.255.255.255
 permit ip 10.20.0.0 0.0.255.255 any
ip nat inside source list ACL-NAT interface GigabitEthernet0/0 overload
ip route 0.0.0.0 0.0.0.0 198.51.100.1

crypto isakmp policy 10
 encr aes 256
 hash sha
 authentication pre-share
 group 5
 lifetime 86400
crypto isakmp key VpnLab#2026 address 203.0.113.2
crypto ipsec transform-set TS-SITES esp-aes 256 esp-sha-hmac
ip access-list extended ACL-VPN
 permit gre host 198.51.100.2 host 203.0.113.2
crypto map CM-SITES 10 ipsec-isakmp
 set peer 203.0.113.2
 set transform-set TS-SITES
 match address ACL-VPN
interface GigabitEthernet0/0
 crypto map CM-SITES

interface Tunnel0
 description VPN vers R-SIEGE (GRE over IPsec)
 ip address 172.16.0.2 255.255.255.252
 ip mtu 1400
 tunnel source GigabitEthernet0/0
 tunnel destination 203.0.113.2
 ip ospf authentication message-digest
 ip ospf message-digest-key 1 md5 OspfLab#2026

router ospf 1
 router-id 3.3.3.3
 passive-interface default
 no passive-interface Tunnel0
 network 10.20.0.0 0.0.255.255 area 0
 network 172.16.0.0 0.0.0.3 area 0
end
write memory
