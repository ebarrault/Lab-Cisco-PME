!  SW-AGENCE — commutateur d'accès Cisco 2960-24TT de l'agence
!  Fa0/1-20 : ADMINISTRATIF (VLAN 10) -> PC-AGENCE1 Fa0/1, PC-AGENCE2 Fa0/2
!  Fa0/24   : poste d'administration (VLAN 99)
!  Gi0/1    : trunk vers R-AGENCE G0/1

enable
configure terminal
hostname SW-AGENCE
no ip domain-lookup
ip domain-name pme.local

service password-encryption
enable secret Cisco#Lab2026!
username admin privilege 15 secret Admin#Lab2026!
banner motd # Acces reserve au personnel autorise. Toute connexion est journalisee. #
no ip http server
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip access-list standard ACL-ADMIN-SSH
 permit 10.10.99.0 0.0.0.255
 permit 10.20.99.0 0.0.0.255
 deny any log
line console 0
 login local
 exec-timeout 5 0
 logging synchronous
line vty 0 15
 login local
 transport input ssh
 access-class ACL-ADMIN-SSH in
 exec-timeout 5 0

vtp mode transparent
vlan 10
 name ADMINISTRATIF
vlan 99
 name MANAGEMENT
vlan 999
 name PARKING-NATIF
spanning-tree mode rapid-pvst

ip dhcp snooping
ip dhcp snooping vlan 10
no ip dhcp snooping information option

interface GigabitEthernet0/1
 description Trunk vers R-AGENCE
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,99,999
 switchport nonegotiate
 ip dhcp snooping trust
 no shutdown

interface range FastEthernet0/1 - 20
 description Postes ADMINISTRATIF
 switchport mode access
 switchport access vlan 10
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown

interface FastEthernet0/24
 description Poste administration
 switchport mode access
 switchport access vlan 99
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown

interface range FastEthernet0/21 - 23 , GigabitEthernet0/2
 description Port inutilise
 switchport mode access
 switchport access vlan 999
 shutdown

interface Vlan1
 no ip address
 shutdown
interface Vlan99
 description Management
 ip address 10.20.99.11 255.255.255.0
 no shutdown
ip default-gateway 10.20.99.1
end
write memory
