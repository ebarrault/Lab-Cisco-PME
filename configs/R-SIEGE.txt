! =====================================================================
!  R-SIEGE — routeur Cisco 2911 du siège (Bayonne)
!  Rôles : sortie Internet (NAT/PAT), serveur DHCP des VLAN du siège,
!          VPN GRE over IPsec vers l'agence, OSPF zone 0
!  Prérequis Packet Tracer pour IPsec :
!    license boot module c2900 technology-package securityk9  -> yes -> write -> reload
! =====================================================================
enable
configure terminal
hostname R-SIEGE
no ip domain-lookup
ip domain-name pme.local

! ---------- Durcissement ----------
service password-encryption
security passwords min-length 10
enable secret Cisco#Lab2026!
username admin privilege 15 secret Admin#Lab2026!
login block-for 120 attempts 3 within 60
banner motd # Acces reserve au personnel autorise. Toute connexion est journalisee. #
no ip http server
logging buffered 16384
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3

ip access-list standard ACL-ADMIN-SSH
 remark Seuls les reseaux d'administration peuvent ouvrir une session SSH
 permit 10.10.99.0 0.0.0.255
 permit 10.20.99.0 0.0.0.255
 deny any log

line console 0
 login local
 exec-timeout 5 0
 logging synchronous
line vty 0 4
 login local
 transport input ssh
 access-class ACL-ADMIN-SSH in
 exec-timeout 5 0
line vty 5 15
 transport input none

! ---------- Interfaces ----------
interface GigabitEthernet0/0
 description WAN - vers ISP
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 no cdp enable
 no shutdown
interface GigabitEthernet0/1
 description LAN - lien route vers SW-CORE-SIEGE Gi0/1
 ip address 10.10.255.1 255.255.255.252
 ip nat inside
 ip ospf authentication message-digest
 ip ospf message-digest-key 1 md5 OspfLab#2026
 no shutdown

! ---------- DHCP (les SVI du coeur relaient vers 10.10.255.1) ----------
ip dhcp excluded-address 10.10.10.1 10.10.10.20
ip dhcp excluded-address 10.10.20.1 10.10.20.20
ip dhcp excluded-address 10.10.40.1 10.10.40.20
ip dhcp pool VLAN10-ADMINISTRATIF
 network 10.10.10.0 255.255.255.0
 default-router 10.10.10.1
 dns-server 10.10.30.10
 domain-name pme.local
ip dhcp pool VLAN20-COMPTA
 network 10.10.20.0 255.255.255.0
 default-router 10.10.20.1
 dns-server 10.10.30.10
 domain-name pme.local
ip dhcp pool VLAN40-INVITES
 network 10.10.40.0 255.255.255.0
 default-router 10.10.40.1
 dns-server 192.0.2.10

! ---------- NAT / PAT ----------
ip access-list extended ACL-NAT
 remark Pas de NAT entre les sites (le trafic inter-sites passe par le tunnel)
 deny ip 10.0.0.0 0.255.255.255 10.0.0.0 0.255.255.255
 permit ip 10.10.0.0 0.0.255.255 any
ip nat inside source list ACL-NAT interface GigabitEthernet0/0 overload
! Publication du serveur web intranet sur l'adresse publique (redirection de port)
ip nat inside source static tcp 10.10.30.30 80 203.0.113.2 80

ip route 0.0.0.0 0.0.0.0 203.0.113.1

! ---------- VPN site à site : GRE protégé par IPsec ----------
crypto isakmp policy 10
 encr aes 256
 hash sha
 authentication pre-share
 group 5
 lifetime 86400
crypto isakmp key VpnLab#2026 address 198.51.100.2
crypto ipsec transform-set TS-SITES esp-aes 256 esp-sha-hmac
ip access-list extended ACL-VPN
 remark Chiffre tout le trafic GRE entre les deux routeurs
 permit gre host 203.0.113.2 host 198.51.100.2
crypto map CM-SITES 10 ipsec-isakmp
 set peer 198.51.100.2
 set transform-set TS-SITES
 match address ACL-VPN
interface GigabitEthernet0/0
 crypto map CM-SITES

interface Tunnel0
 description VPN vers R-AGENCE (GRE over IPsec)
 ip address 172.16.0.1 255.255.255.252
 ip mtu 1400
 tunnel source GigabitEthernet0/0
 tunnel destination 198.51.100.2
 ip ospf authentication message-digest
 ip ospf message-digest-key 1 md5 OspfLab#2026

! ---------- OSPF ----------
router ospf 1
 router-id 1.1.1.1
 passive-interface default
 no passive-interface GigabitEthernet0/1
 no passive-interface Tunnel0
 network 10.10.255.0 0.0.0.3 area 0
 network 172.16.0.0 0.0.0.3 area 0
 default-information originate
end
write memory
