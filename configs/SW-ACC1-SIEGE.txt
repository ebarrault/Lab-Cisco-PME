! =====================================================================
!  SW-ACC1-SIEGE — commutateur d'accès Cisco 2960-24TT
!  Fa0/1-9   : postes ADMINISTRATIF (VLAN 10)  -> PC-ADM1 sur Fa0/1
!  Fa0/10-12 : serveurs (VLAN 30)               -> SRV-AD, SRV-FICHIERS, SRV-WEB
!  Fa0/24    : poste d'administration IT (VLAN 99)
!  Gi0/1     : trunk vers SW-CORE-SIEGE Fa0/1
! =====================================================================
enable
configure terminal
hostname SW-ACC1-SIEGE
no ip domain-lookup
ip domain-name pme.local

service password-encryption
enable secret Cisco#Lab2026!
username admin privilege 15 secret Admin#Lab2026!
banner motd # Acces reserve au personnel autorise. Toute connexion est journalisee. #
no ip http server
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip access-list standard ACL-ADMIN-SSH
 permit 10.10.99.0 0.0.0.255
 permit 10.20.99.0 0.0.0.255
 deny any log
line console 0
 login local
 exec-timeout 5 0
 logging synchronous
line vty 0 15
 login local
 transport input ssh
 access-class ACL-ADMIN-SSH in
 exec-timeout 5 0

vtp mode transparent
vlan 10
 name ADMINISTRATIF
vlan 30
 name SERVEURS
vlan 99
 name MANAGEMENT
vlan 999
 name PARKING-NATIF
spanning-tree mode rapid-pvst

! ---------- Protection contre les faux serveurs DHCP ----------
ip dhcp snooping
ip dhcp snooping vlan 10
no ip dhcp snooping information option

interface GigabitEthernet0/1
 description Trunk vers SW-CORE-SIEGE
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,30,99
 switchport nonegotiate
 ip dhcp snooping trust
 no shutdown

interface range FastEthernet0/1 - 9
 description Postes ADMINISTRATIF
 switchport mode access
 switchport access vlan 10
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown

interface range FastEthernet0/10 - 12
 description Serveurs
 switchport mode access
 switchport access vlan 30
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown

interface FastEthernet0/24
 description Poste administration IT
 switchport mode access
 switchport access vlan 99
 switchport port-security
 switchport port-security mac-address sticky
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown

interface range FastEthernet0/13 - 23 , GigabitEthernet0/2
 description Port inutilise
 switchport mode access
 switchport access vlan 999
 shutdown

interface Vlan1
 no ip address
 shutdown
interface Vlan99
 description Management
 ip address 10.10.99.11 255.255.255.0
 no shutdown
ip default-gateway 10.10.99.1
end
write memory
