! =====================================================================
!  SW-CORE-SIEGE — commutateur de niveau 3 Cisco 3560-24PS (coeur du siège)
!  Rôles : routage inter-VLAN (SVI), racine spanning-tree, relais DHCP,
!          ACL invités et serveurs, OSPF zone 0
!  Fa0/1 -> SW-ACC1-SIEGE Gi0/1 (trunk) | Fa0/2 -> SW-ACC2-SIEGE Gi0/1 (trunk)
!  Gi0/1 -> R-SIEGE G0/1 (port routé)
! =====================================================================
enable
configure terminal
hostname SW-CORE-SIEGE
no ip domain-lookup
ip domain-name pme.local
ip routing

! ---------- Durcissement ----------
service password-encryption
enable secret Cisco#Lab2026!
username admin privilege 15 secret Admin#Lab2026!
banner motd # Acces reserve au personnel autorise. Toute connexion est journalisee. #
no ip http server
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip access-list standard ACL-ADMIN-SSH
 permit 10.10.99.0 0.0.0.255
 permit 10.20.99.0 0.0.0.255
 deny any log
line console 0
 login local
 exec-timeout 5 0
 logging synchronous
line vty 0 15
 login local
 transport input ssh
 access-class ACL-ADMIN-SSH in
 exec-timeout 5 0

! ---------- VLAN ----------
vtp mode transparent
vlan 10
 name ADMINISTRATIF
vlan 20
 name COMPTABILITE
vlan 30
 name SERVEURS
vlan 40
 name WIFI-INVITES
vlan 99
 name MANAGEMENT
vlan 999
 name PARKING-NATIF

spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,99 root primary

! ---------- Trunks vers les commutateurs d'accès ----------
interface FastEthernet0/1
 description Trunk vers SW-ACC1-SIEGE
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,30,99
 switchport nonegotiate
 no shutdown
interface FastEthernet0/2
 description Trunk vers SW-ACC2-SIEGE
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,40,99
 switchport nonegotiate
 no shutdown
interface range FastEthernet0/3 - 24
 description Port inutilise
 switchport mode access
 switchport access vlan 999
 shutdown
interface GigabitEthernet0/2
 description Port inutilise
 switchport mode access
 switchport access vlan 999
 shutdown

! ---------- Lien routé vers le routeur ----------
interface GigabitEthernet0/1
 description Lien route vers R-SIEGE G0/1
 no switchport
 ip address 10.10.255.2 255.255.255.252
 ip ospf authentication message-digest
 ip ospf message-digest-key 1 md5 OspfLab#2026
 no shutdown

! ---------- ACL ----------
ip access-list extended ACL-INVITES-IN
 remark Invites : DHCP et Internet uniquement, aucun acces aux reseaux internes
 permit udp any any eq bootps
 deny ip any 10.0.0.0 0.255.255.255
 deny ip any 172.16.0.0 0.15.255.255
 deny ip any 192.168.0.0 0.0.255.255
 permit ip 10.10.40.0 0.0.0.255 any

ip access-list extended ACL-SERVEURS-OUT
 remark Trafic autorise vers le VLAN serveurs (applique en sortie de la SVI 30)
 permit ip 10.10.99.0 0.0.0.255 10.10.30.0 0.0.0.255
 permit ip 10.20.99.0 0.0.0.255 10.10.30.0 0.0.0.255
 permit tcp 10.10.10.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 80
 permit tcp 10.10.10.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 443
 permit tcp 10.10.10.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 445
 permit udp 10.10.10.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 53
 permit tcp 10.10.20.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 80
 permit tcp 10.10.20.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 443
 permit tcp 10.10.20.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 445
 permit udp 10.10.20.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 53
 permit tcp 10.20.10.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 80
 permit tcp 10.20.10.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 443
 permit tcp 10.20.10.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 445
 permit udp 10.20.10.0 0.0.0.255 10.10.30.0 0.0.0.255 eq 53
 permit icmp 10.0.0.0 0.255.255.255 10.10.30.0 0.0.0.255 echo
 remark Retours des connexions initiees par les serveurs
 permit tcp any 10.10.30.0 0.0.0.255 established
 permit udp any eq 53 10.10.30.0 0.0.0.255
 permit icmp any 10.10.30.0 0.0.0.255 echo-reply
 remark Site web publie sur Internet par NAT statique
 permit tcp any host 10.10.30.30 eq 80
 deny ip any any log

! ---------- SVI (passerelles des VLAN) ----------
interface Vlan10
 description Passerelle ADMINISTRATIF
 ip address 10.10.10.1 255.255.255.0
 ip helper-address 10.10.255.1
 no shutdown
interface Vlan20
 description Passerelle COMPTABILITE
 ip address 10.10.20.1 255.255.255.0
 ip helper-address 10.10.255.1
 no shutdown
interface Vlan30
 description Passerelle SERVEURS (adresses fixes)
 ip address 10.10.30.1 255.255.255.0
 ip access-group ACL-SERVEURS-OUT out
 no shutdown
interface Vlan40
 description Passerelle WIFI-INVITES
 ip address 10.10.40.1 255.255.255.0
 ip helper-address 10.10.255.1
 ip access-group ACL-INVITES-IN in
 no shutdown
interface Vlan99
 description Management
 ip address 10.10.99.1 255.255.255.0
 no shutdown

! ---------- OSPF ----------
router ospf 1
 router-id 2.2.2.2
 passive-interface default
 no passive-interface GigabitEthernet0/1
 network 10.10.0.0 0.0.255.255 area 0
end
write memory
